\DTLnewdb{kitDB}
% ------------------------------------------------------------------------------------------------- %
% ################################################################################################# %
% -------------------------------------- Add  Database Entry -------------------------------------- %
\NewDocumentCommand{\AddKitDBEntry}{ mmmmmm }{%
	\DTLnewrow{kitDB}%
	\DTLnewdbentry{kitDB}{Name}{#1}%
	\DTLnewdbentry{kitDB}{WeaponImplement}{#2}%
	\DTLnewdbentry{kitDB}{ArmorWard}{#3}%
	
	\SetKitBoni#4
	
	\SetDamageBoni#5
	
	\DTLnewdbentry{kitDB}{Effects}{#6}%
}%
% ------------------------------------------------------------------------------------------------- %
% +++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++ %
% #-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-# Set  Kit-Boni #-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-# %
\NewDocumentCommand{\SetKitBoni}{ mmmmmm }{%
	\DTLnewdbentry{kitDB}{Speed}{#1}%
	\DTLnewdbentry{kitDB}{Disengage}{#2}%
	\DTLnewdbentry{kitDB}{Stamina}{#3}%
	\DTLnewdbentry{kitDB}{Stability}{#4}%
	\DTLnewdbentry{kitDB}{MeleeDistance}{#5}%
	\DTLnewdbentry{kitDB}{RangedDistance}{#6}%
}%
% ------------------------------------------------------------------------------------------------- %
% +++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++++ %
% #-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#  Set Damage Boni  #-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-#-# %
\NewDocumentCommand{\SetDamageBoni}{ mm }{%
	\DTLnewdbentry{kitDB}{MeleeDamageBonus}{#1}%
	\DTLnewdbentry{kitDB}{RangedDamageBonus}{#2}%
}%
