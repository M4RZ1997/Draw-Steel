% ------------------------------------------- Variables ------------------------------------------- %
\DTLnewdb{actionDB}

\newcommand{\CategorySymbolValue}{}
\newcommand{\CategorySymbol}[1]{\renewcommand{\CategorySymbolValue}{#1}}
\newbool{FoundEntry}
% ------------------------------------------------------------------------------------------------- %
% ################################################################################################# %
% -------------------------------------- Add  Database Entry -------------------------------------- %
\NewDocumentCommand{\AddActionDBEntry}{ O{} mmmmmmmm }{%
	\DTLnewrow{actionDB}%
	
	\DTLnewdbentry{actionDB}{PanelSize}{#1}%
	
	\DTLnewdbentry{actionDB}{Type}{#2}%
	\DTLnewdbentry{actionDB}{Name}{#3}%
	\DTLnewdbentry{actionDB}{Cost}{#4}%
	\DTLnewdbentry{actionDB}{Description}{#5}%
	\DTLnewdbentry{actionDB}{Category}{#6}%
	\DTLnewdbentry{actionDB}{Range}{#7}%
	\DTLnewdbentry{actionDB}{Targets}{#8}%
	\DTLnewdbentry{actionDB}{Textbox}{#9}%
}%
% ------------------------------------------------------------------------------------------------- %
% ################################################################################################# %
% ----------------------------------- Action  Short Information ----------------------------------- %
\NewDocumentCommand{\GetActionShortInfo}{ m }{%
	\boolfalse{FoundEntry}
	\DTLforeach*{actionDB}{%
		\thisType			= Type,%
		\thisName			= Name,%
		\thisDescription	= Description,%
		\thisCategory		= Category%
	}{%
		\ifthenelse{\equal{#1}{\thisName}}{%
			\booltrue{FoundEntry}%
			\SetCategorySymbol{\thisType}{\thisCategory}%
			\textbf{\CategorySymbolValue~\thisName}\hfill\GetActionType{\thisType}%
			\ifthenelse{\equal{\thisDescription}{}}{}{\\\textit{\thisDescription}}%
		}{}%
	}%
	\ifbool{FoundEntry}{}{#1}%
}%
% ------------------------------------------------------------------------------------------------- %
% ################################################################################################# %
% -------------------------------------- Set Category Symbol -------------------------------------- %
\NewDocumentCommand{\SetCategorySymbol}{ m m }{%
	\IfSubStr{#1}{Triggered}{%
		\CategorySymbol{{\dsglyphs !}}%
	}{%
		\IfSubStr{#2}{Magic}{%
			\CategorySymbol{{\dsglyphs *}}%
		}{%
			\IfSubStr{#2}{Melee}{%
				\IfSubStr{#2}{Ranged}{%
					\CategorySymbol{{\dsglyphs =}}%
				}{%
					\CategorySymbol{{\dsglyphs <}}
				}%
			}{%
				\IfSubStr{#2}{Ranged}{\CategorySymbol{{\dsglyphs >}}}{}%
			}%
		}%
	}%
}
