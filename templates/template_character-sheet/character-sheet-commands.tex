\newcommand{\CalculateKitInformation}{%
	\DTLforeach*{kitDB}{%
		\thisKitName			= Name,%
		\thisWeaponImplement	= WeaponImplement,%
		\thisArmorWard			= ArmorWard,%
		\thisSpeed				= Speed,%
		\thisDisengage			= Disengage,%
		\thisStamina			= Stamina,%
		\thisStability			= Stability,%
		\thisMeleeDistance		= MeleeDistance,%
		\thisRangedDistance		= RangedDistance,%
		\thisMeleeDamageBonus	= MeleeDamageBonus,%
		\thisRangedDamageBonus	= RangedDamageBonus,%
		\thisEffects			= Effects%
	}{%
		\ifthenelse{\equal{\KitsNameValue}{}}{%
			\KitsName{\thisKitName}%
		}{%
			\ifthenelse{\equal{\thisKitName}{}}{}{%
				\g@addto@macro\KitsNameValue{~\&~\thisKitName}%
			}%
		}%
		
		\ifthenelse{\equal{\WeaponImplementValue}{}}{%
			\WeaponImplement{\thisWeaponImplement}%
		}{%
			\ifthenelse{\equal{\thisWeaponImplement}{}}{}{%
				\g@addto@macro\WeaponImplementValue{,~\thisWeaponImplement}%
			}%
		}%
		
		\ifthenelse{\equal{\ArmorWardValue}{}}{%
			\ArmorWard{\thisArmorWard}%
		}{%
			\ifthenelse{\equal{\thisArmorWard}{}}{}{%
				\g@addto@macro\ArmorWardValue{,~\thisArmorWard}%
			}%
		}%
		
		\ifthenelse{\equal{\thisSpeed}{}}{}{%
			\ifthenelse{\thisSpeed>\SpeedKitBonusValue}{%
				\SpeedKitBonus{\thisSpeed}
			}{}%
		}
		\ifthenelse{\equal{\thisDisengage}{}}{}{%
			\ifthenelse{\thisDisengage>\DisengageKitBonusValue}{%
				\DisengageKitBonus{\thisDisengage}
			}{}%
		}
		\ifthenelse{\equal{\thisStamina}{}}{}{%
			\ifthenelse{\thisStamina>\StaminaKitBonusValue}{%
				\StaminaKitBonus{\thisStamina}
			}{}%
		}
		\ifthenelse{\equal{\thisStability}{}}{}{%
			\ifthenelse{\thisStability>\StabilityKitBonusValue}{%
				\StabilityKitBonus{\thisStability}
			}{}%
		}
		\ifthenelse{\equal{\thisMeleeDistance}{}}{}{%
			\ifthenelse{\thisMeleeDistance>\MeleeDistanceKitBonusValue}{%
				\MeleeDistanceKitBonus{\thisMeleeDistance}
			}{}%
		}
		\ifthenelse{\equal{\thisRangedDistance}{}}{}{%
			\ifthenelse{\thisRangedDistance>\RangedDistanceKitBonusValue}{%
				\RangedDistanceKitBonus{\thisRangedDistance}
			}{}%
		}
		
		\ifthenelse{\equal{\thisMeleeDamageBonus}{}}{}{%
			\SetMeleeDamageBonus{\thisMeleeDamageBonus}%
		}%
		\ifthenelse{\equal{\thisRangedDamageBonus}{}}{}{%
			\SetRangedDamageBonus{\thisRangedDamageBonus}%
		}%
		
		\foreach \action in \thisEffects {%
			\g@addto@macro\KitEffectsValue{\GetActionShortInfo{\action}\\\vskip -5pt}
			\KitEffects{\KitEffectsValue}
		}%
	}%
}%

\newcommand{\GetActionType}[1]{%
	\str_case_e:nnF {#1}
	{
		{Main~Action}
		{
			\textcolor{actionpanelmain}{Main~Action}
		}
		
		{Heroic~Ability}
		{
			\textcolor{actionpanelmain}{Main~Action}
		}
		
		{Signature~Ability}
		{
			\textcolor{actionpanelmain}{Main~Action}
		}

		{Maneuver}
		{
			\textcolor{actionpanelmaneuver}{Maneuver}
		}

		{Triggered~Action}
		{
			\textcolor{actionpaneltriggered}{Triggered~Action}
		}

		{Free~Triggered~Action}
		{
			\textcolor{actionpaneltriggered}{Free~Triggered~Action}
		}

		{Free~Strike}
		{
			\textcolor{actionpanelfreestrike}{Free~Strike}
		}
	}
	{
		\IfSubStr{#1}{Free~Triggered~Action}{
			\textcolor{actionpaneltriggered}{Free~Triggered~Action}%
		}{%
			\PackageError{character-sheet-commands}
				{
					Unknown~Action~Panel~type~'#1'
				}
				{
					Valid~action~types~are~Main~Action,~Maneuver,~Triggered~Action,~Free~Triggered~Action,~and~Free~ Strike.~For~Heroic~or~Signature~abilities~use,~for~example,~'Heroic~Ability/Triggered~Action'~or~'Signature~Ability/Main~Action'.
				}
			}%
	}
}%
